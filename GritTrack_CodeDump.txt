        Quiz,
        Test,
        Project,
        Reading,
        Other
    }

        public int ID { get; set; }

        [JsonIgnore]
        public int CourseID { get; set; }

        public string Title { get; set; } = string.Empty;
        public AssignmentType Type { get; set; }
        public DateTime DueDate { get; set; }
        public double PointsPossible { get; set; }
        public double? PointsEarned { get; set; }
        public bool IsCompleted { get; set; }
        public double EstimatedHours { get; set; } = 1.0;
        public string Notes { get; set; } = string.Empty;

        [JsonIgnore]
        public bool IsOverdue => !IsCompleted && DueDate.Date < DateTime.Today;

        [JsonIgnore]
        public bool IsDueSoon => !IsCompleted && !IsOverdue
            && (DueDate.Date - DateTime.Today).TotalDays <= 3;

        public override string ToString() => $"{Title} ({Type}) due {DueDate:MM/dd}";
    }

        {
            if (query.TryGetValue(QueryKeys.EditCourse, out var value) && value is Course course)
                Load(course);
        }

        // called when navigating here to edit an existing course (e.g. bumping
        // a grade after extra credit) instead of adding a brand new one â€”
        // pre-fills the form and flips the title/button text to match
        public void Load(Course course)
        {
            _editingCourseId = course.ID;
            Name = course.Name;
            CourseCode = course.CourseCode;
            Instructor = course.Instructor ?? string.Empty;
            Credits = course.Credits.ToString();
            LetterGrade = string.IsNullOrWhiteSpace(course.CurrentGrade) ? LetterGrades[0] : course.CurrentGrade;
            Title = "Edit Course";
            SaveButtonText = "Save Changes";
            OnPropertyChanged(nameof(IsEditing));
        }

        private bool Validate(out double credits)
        {
            credits = 0;

            // Name is the only field that's actually required â€” Course Code,
            // Instructor and Credits (defaulted to 3) are all optional so this
            // doesn't turn into a form nobody can finish
            if (string.IsNullOrWhiteSpace(Name))
            {
                ValidationMessage = "Course name is required.";
                return false;
            }

            if (!string.IsNullOrWhiteSpace(Credits) && (!double.TryParse(Credits, out credits) || credits < 0))
            {
                ValidationMessage = "Credits has to be a positive number.";
                return false;
            }

            if (string.IsNullOrWhiteSpace(Credits))
                credits = 0;

            ValidationMessage = string.Empty;
            return true;
        }

        private async Task SaveAsync()
        {
            if (IsSaving || !Validate(out var credits))
                return;

            // only the basic "is the name filled in" check happens here. the
            // API has the real rules (code required, credits 0-6, lengths,
            // duplicate codes) and we show whatever it says
            var course = new Course
            {
                // 0 = brand new, the API picks the ID. editing keeps the original
                // ID so the API updates the right course
                ID = _editingCourseId ?? 0,
                Name = Name.Trim(),
                CourseCode = (CourseCode ?? string.Empty).Trim(),
                Instructor = (Instructor ?? string.Empty).Trim(),
                Credits = credits,
                CurrentGrade = LetterGrade
            };

            IsSaving = true;
            CanTryAgain = false;
            ValidationMessage = string.Empty;

            try
            {
                // if the API says "log in first", this opens the login page and,
                // once they sign in, sends the same save again
                var saved = await _authService.RunWriteAsync(() => _courseRepository.SaveCourseAsync(course));

                // "SavedCourse" is the label CourseListPageModel looks for. the
                // course is already saved, the list just needs to show it.
                // adding got here from the list (one level up); editing got here from
                // detail (two levels up: list -> detail -> here), so pop back to
                // the list directly instead of landing on the now-stale detail page
                var route = IsEditing ? Routes.BackTwice : Routes.Back;
                await _navigation.GoToAsync(route, new Dictionary<string, object> { { QueryKeys.SavedCourse, saved } });
            }
            catch (ApiValidationException exception)
            {
                // 400 or 409: the form stays open so they can fix it
                ValidationMessage = string.Join(Environment.NewLine, exception.Messages);
            }
            catch (ApiRequestException exception)
            {
                ValidationMessage = exception.Message;
                CanTryAgain = true;
            }
            catch (LoginCancelledException exception)
            {
                ValidationMessage = exception.Message;
            }
            finally
            {
                IsSaving = false;
            }
        }
    }
}

using System.ComponentModel;
using System.Runtime.CompilerServices;

namespace GritTrack.PageModels
{
    /// <summary>
    /// Hand-rolled MVVM base class â€” no source-gen shortcuts.
    /// Every PageModel inherits this so the View can bind to properties
    /// and get notified when they change.
    /// </summary>
    public class BaseViewModel : INotifyPropertyChanged
    {
        public event PropertyChangedEventHandler? PropertyChanged;

        /// <summary>
        /// Sets the backing field and raises PropertyChanged ONLY if the
        /// value actually changed. This is the pattern from the "What is
        /// SetProperty" lesson â€” ref lets us write directly into the
        /// caller's backing field instead of returning a copy.
        /// </summary>
        protected bool SetProperty<T>(ref T backingStore, T value, [CallerMemberName] string? propertyName = null)
        {
            if (EqualityComparer<T>.Default.Equals(backingStore, value))
                return false;

            backingStore = value;
            OnPropertyChanged(propertyName);
            return true;
        }

        protected void OnPropertyChanged([CallerMemberName] string? propertyName = null)
        {
            PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(propertyName));
        }
    }
}

        {
            _navigation = navigation;
            _courseRepository = courseRepository;
            _authService = authService;

            GoBackCommand = new RelayCommand(() => _ = _navigation.GoToAsync(Routes.Back));
            EditCommand = new RelayCommand(() => _ = _navigation.GoToAsync(
                Routes.AddCourse,
                new Dictionary<string, object> { { QueryKeys.EditCourse, Course } }));
            DeleteCommand = new RelayCommand(() => IsConfirmingDelete = true);

            // also what the Try Again button runs after a failed delete
            ConfirmDeleteCommand = new RelayCommand(() =>
            {
                IsConfirmingDelete = false;
                _ = DeleteAsync();
            });
            CancelDeleteCommand = new RelayCommand(() => IsConfirmingDelete = false);
        }

        {
            if (IsBusy)
                return;

            IsBusy = true;
            ErrorMessage = string.Empty;

                });

                    Routes.BackTwice,
                    new Dictionary<string, object> { { QueryKeys.DeletedCourse, Course } });
            }
            catch (Exception exception) when (exception is ApiRequestException or LoginCancelledException)
            {
                ErrorMessage = exception.Message;
            }
            finally
            {
                IsBusy = false;
            }
        }
    }

                StartingGpaValidationMessage = "GPA needs to be a number between 0.0 and 4.0";
                return;

            {
                StartingGpaValidationMessage = "Credits needs to be a number greater than 0";
                return;
            }

            _startingGpa = gpa;
            _startingCredits = credits;

        // rebuilds FilteredCourses from Courses using the search box. matches
        // the course name, the course code or the instructor
        private void ApplyFilter()
        {
            FilteredCourses.Clear();

            var query = SearchQuery?.Trim().ToLowerInvariant() ?? string.Empty;

            foreach (var course in Courses)
            {
                if (string.IsNullOrEmpty(query) ||

        // lists every assignment that isn't graded yet, each starting at 90%
        private void BuildSimulationItems()
        {
            SimulatedAssignments.Clear();

            foreach (var course in Courses)
            {
                var ungradedAssignments = course.Assignments
                    .Where(assignment => !assignment.IsCompleted || !assignment.PointsEarned.HasValue);

                foreach (var assignment in ungradedAssignments)
                {
                    SimulatedAssignments.Add(new SimulatedAssignmentItem
                    {
                        AssignmentId = assignment.ID,
                        CourseId = course.ID,
                        CourseCode = course.CourseCode,
                        Title = assignment.Title,
                        PointsPossible = assignment.PointsPossible,
                        ExpectedPoints = Math.Round(assignment.PointsPossible * 0.9),
                        OnScoreChanged = RecalculateSimulatedGpa
                    });
                }
            }

            RecalculateSimulatedGpa();
        }

        // works out the GPA as if every simulated score really happens. graded
        // work counts as it is, simulated work gets added on top, then each
        // course's percent turns into a letter grade like the real calculation
        private void RecalculateSimulatedGpa()
        {
            var simulatedCourses = new List<Course>();

            foreach (var course in Courses)
            {
                var gradedAssignments = course.Assignments
                    .Where(assignment => assignment.IsCompleted && assignment.PointsEarned.HasValue && assignment.PointsPossible > 0)
                    .ToList();

                var pointsEarned = gradedAssignments.Sum(assignment => assignment.PointsEarned!.Value);
                var pointsPossible = gradedAssignments.Sum(assignment => assignment.PointsPossible);

                foreach (var simulatedAssignment in SimulatedAssignments.Where(item => item.CourseId == course.ID))
                {
                    pointsEarned += simulatedAssignment.ExpectedPoints;
                    pointsPossible += simulatedAssignment.PointsPossible;
                }

                if (pointsPossible > 0)
                {
                    var percent = (pointsEarned / pointsPossible) * 100.0;
                    var letterGrade = _gpaCalculator.GetLetterGradeFromPercent(percent);

                    simulatedCourses.Add(new Course
                    {
                        ID = course.ID,
                        Name = course.Name,
                        Credits = course.Credits,
                        CurrentGrade = letterGrade,
                        GradePoints = _gpaCalculator.GetGradePoints(letterGrade)
                    });
                }
                else
                {
                    // nothing to go on, so keep the course's real grade
                    simulatedCourses.Add(course);
                }
            }

            ProjectedGpa = _startingCredits > 0
                ? _gpaCalculator.CalculateGpa(simulatedCourses, _startingGpa, _startingCredits)
                : _gpaCalculator.CalculateGpa(simulatedCourses);

            ProjectedStatus = _gpaCalculator.GetStatus(ProjectedGpa, PtkGoodStandingGpa, PtkInitiationGpa);
        }

        private void RecalculateGpa()
        {
            // Courses just changed, so the search results need a refresh too
            ApplyFilter();

            CurrentGpa = _startingCredits > 0
                ? _gpaCalculator.CalculateGpa(Courses, _startingGpa, _startingCredits)
                : _gpaCalculator.CalculateGpa(Courses);

            Status = _gpaCalculator.GetStatus(CurrentGpa, PtkGoodStandingGpa, PtkInitiationGpa);

            // the actual dynamic-resource trigger â€” every time GPA changes,
            // the app's accent color updates to match how you're doing
            _themeService.ApplyGpaStatusTheme(Status);

            // keep the open simulator in step with the new real numbers
            if (IsSimulatorOpen)
                RecalculateSimulatedGpa();
        }

namespace GritTrack.PageModels

    /// <summary>
    /// Hand-rolled ICommand implementation â€” no CommunityToolkit
    /// [RelayCommand] source-gen. This is what the "Commands and
    /// Converters" lesson describes: Execute runs the action,
    /// CanExecute controls whether the bound button is enabled.
    /// </summary>
    public class RelayCommand : ICommand
    {
        private readonly Action _execute;
        private readonly Func<bool>? _canExecute;

        public RelayCommand(Action execute, Func<bool>? canExecute = null)
        {
            _execute = execute ?? throw new ArgumentNullException(nameof(execute));
            _canExecute = canExecute;
        }

        public event EventHandler? CanExecuteChanged;

        public bool CanExecute(object? parameter) => _canExecute?.Invoke() ?? true;

        public void Execute(object? parameter) => _execute();

        public void RaiseCanExecuteChanged() =>
            CanExecuteChanged?.Invoke(this, EventArgs.Empty);
    }

    /// <summary>
    /// Generic version for commands that need a parameter
    /// (e.g. "which course did you tap").
    /// </summary>
    public class RelayCommand<T> : ICommand
    {

        public RelayCommand(Action<T?> execute, Func<T?, bool>? canExecute = null)
        {
            _execute = execute ?? throw new ArgumentNullException(nameof(execute));
            _canExecute = canExecute;

namespace GritTrack.PageModels
{
    // one upcoming assignment in the what-if GPA simulator. the slider on the
    // page moves ExpectedPoints, and OnScoreChanged tells the list page to
    // redo the projected GPA
    public class SimulatedAssignmentItem : BaseViewModel
    {
        public int AssignmentId { get; set; }
        public int CourseId { get; set; }
        public string CourseCode { get; set; } = string.Empty;
        public string Title { get; set; } = string.Empty;
        public double PointsPossible { get; set; }

        private double _expectedPoints;
        public double ExpectedPoints
        {
            get => _expectedPoints;
            set
            {
                // can't score above what the assignment is worth
                var clampedPoints = Math.Clamp(value, 0, PointsPossible);
                if (SetProperty(ref _expectedPoints, clampedPoints))
                {
                    OnPropertyChanged(nameof(ExpectedPercent));
                    OnScoreChanged?.Invoke();
                }
            }
        }

        public double ExpectedPercent => PointsPossible > 0 ? Math.Round((ExpectedPoints / PointsPossible) * 100) : 0;

        public Action? OnScoreChanged { get; set; }
    }

    public class ApiCourseRepository : ICourseRepository
    {
        private readonly HttpClient _client;

        // the HttpClient comes from the factory in MauiProgram, already pointed
        // at the API and wired to add the login token on writes
        public ApiCourseRepository(HttpClient client)
        {
            _client = client;
        }

        public async Task<List<Course>> GetCoursesAsync()
        {
            using var request = new HttpRequestMessage(HttpMethod.Get, "api/courses");
            using var response = await ApiResponseReader.SendAsync(_client, request);
            await ApiResponseReader.EnsureSuccessAsync(response);

            var courses = await response.Content.ReadFromJsonAsync<List<CourseResponse>>(ApiResponseReader.JsonOptions);
            return courses?.Select(course => course.ToCourse()).ToList() ?? new List<Course>();
        }

        // ID 0 means "brand new" (POST); any other ID is an edit (PUT).
        // gives back the course as the server saved it, with its real ID
        public async Task<Course> SaveCourseAsync(Course course)
        {
            var isNew = course.ID == 0;

            using var request = new HttpRequestMessage(
                isNew ? HttpMethod.Post : HttpMethod.Put,
                isNew ? "api/courses" : $"api/courses/{course.ID}")
            {
                Content = JsonContent.Create(CourseRequest.From(course), options: ApiResponseReader.JsonOptions)
            };

            using var response = await ApiResponseReader.SendAsync(_client, request);
            await ApiResponseReader.EnsureSuccessAsync(response);

            var saved = await response.Content.ReadFromJsonAsync<CourseResponse>(ApiResponseReader.JsonOptions);
            return saved?.ToCourse() ?? throw new ApiRequestException("The server answered, but not with the saved course. Try again.");
        }

        public async Task DeleteCourseAsync(int courseId)
        {
            using var request = new HttpRequestMessage(HttpMethod.Delete, $"api/courses/{courseId}");

using System.Net;
using System.Text.Json;

namespace GritTrack.Services.Api
{
    // turns a failed HTTP answer into one of our own exceptions, and wraps the
    // "couldn't reach the server" cases. shared by the course and login calls
    public static class ApiResponseReader
    {
        public static readonly JsonSerializerOptions JsonOptions = new(JsonSerializerDefaults.Web);

        // sends the request. a dead network, a refused connection or a timeout
        // all become the same friendly ApiRequestException
        public static async Task<HttpResponseMessage> SendAsync(HttpClient client, HttpRequestMessage request)
        {
            try
            {
                return await client.SendAsync(request);
            }
            catch (Exception exception) when (exception is HttpRequestException or TaskCanceledException)
            {
                throw new ApiRequestException("Couldn't reach the GritTrack server. Check that the API is running and try again.");
            }
        }

        // does nothing for a 2xx answer, otherwise throws the matching exception
        public static async Task EnsureSuccessAsync(HttpResponseMessage response)
        {
            if (response.IsSuccessStatusCode)
                return;

            switch (response.StatusCode)
            {
                case HttpStatusCode.Unauthorized:
                    throw new AuthenticationRequiredException();

                case HttpStatusCode.BadRequest:
                    throw new ApiValidationException(await ReadProblemMessagesAsync(response, "Some of the course details aren't valid."));

                case HttpStatusCode.Conflict:
                    throw new ApiValidationException(await ReadProblemMessagesAsync(response, "That course code is already in use."));

                case HttpStatusCode.NotFound:
                    throw new ApiRequestException("That course no longer exists. Pull down to refresh the list.");

                default:
                    throw new ApiRequestException($"The server had a problem (error {(int)response.StatusCode}). Try again in a moment.");
            }
        }

        // reads the ProblemDetails body. a 400 has an "errors" object (field name
        // to a list of messages); a 409 has a "detail" sentence
        private static async Task<IReadOnlyList<string>> ReadProblemMessagesAsync(HttpResponseMessage response, string fallbackMessage)
        {
            try
            {
                using var document = await JsonDocument.ParseAsync(await response.Content.ReadAsStreamAsync());
                var messages = new List<string>();

                if (document.RootElement.TryGetProperty("errors", out var errors) && errors.ValueKind == JsonValueKind.Object)
                {
                    foreach (var field in errors.EnumerateObject())
                    {
                        foreach (var message in field.Value.EnumerateArray())
                        {
                            var text = message.GetString();
                            if (!string.IsNullOrWhiteSpace(text))
                                messages.Add(text);
                        }
                    }
                }

                if (messages.Count == 0 && document.RootElement.TryGetProperty("detail", out var detail))
                {
                    var text = detail.GetString();
                    if (!string.IsNullOrWhiteSpace(text))
                        messages.Add(text);
                }

                if (messages.Count > 0)
                    return messages;
            }
            catch (Exception exception) when (exception is JsonException or InvalidOperationException)
            {
                // the body wasn't the shape we expected: fall through to the generic line
            }

            return new[] { fallbackMessage };
        }
    }
}

namespace GritTrack.Services.Api
{
    // keeps the login token in SecureStorage (the phone/PC's protected store),
    // never in plain preferences or a file. the token is a "key card" the API
    // gave us at login; we show it on every write
    public class AuthService : IAuthService
    {
        public const string HttpClientName = "GritTrackAuth";

        private const string TokenKey = "grittrack_token";
        private const string ExpiresKey = "grittrack_token_expires";
        private const string UserNameKey = "grittrack_user";

        private readonly IHttpClientFactory _httpClientFactory;
        private readonly INavigationService _navigation;

        // a copy kept in memory too, so the app still works if secure storage
        // isn't available on this machine
        private string? _token;
        private DateTime _expiresAtUtc;
        private TaskCompletionSource<bool>? _pendingLogin;

        public AuthService(IHttpClientFactory httpClientFactory, INavigationService navigation)
        {
            _httpClientFactory = httpClientFactory;
            _navigation = navigation;
        }

        public event EventHandler? SignInChanged;

        public bool IsSignedIn => _token is not null && _expiresAtUtc > DateTime.UtcNow;

        public string UserName { get; private set; } = string.Empty;

        public async Task RestoreAsync()
        {
            try
            {
                var token = await SecureStorage.Default.GetAsync(TokenKey);
                var expires = await SecureStorage.Default.GetAsync(ExpiresKey);
                var userName = await SecureStorage.Default.GetAsync(UserNameKey);

                if (token is not null && DateTime.TryParse(expires, null, System.Globalization.DateTimeStyles.RoundtripKind, out var expiresAtUtc))
                {
                    _token = token;
                    _expiresAtUtc = expiresAtUtc;
                    UserName = userName ?? string.Empty;
                }
            }
            catch (Exception)
            {
                // secure storage can be missing or locked. start signed out
            }

            if (!IsSignedIn)
                await ClearAsync();
            else
                SignInChanged?.Invoke(this, EventArgs.Empty);
        }

        public Task<string?> GetTokenAsync() => Task.FromResult(IsSignedIn ? _token : null);

        public async Task SignInAsync(string userName, string password)
        {
            var client = _httpClientFactory.CreateClient(HttpClientName);
            using var request = new HttpRequestMessage(HttpMethod.Post, "api/auth/login")
            {
                Content = JsonContent.Create(new LoginRequest { UserName = userName, Password = password }, options: ApiResponseReader.JsonOptions)
            };

            using var response = await ApiResponseReader.SendAsync(client, request);

            // 401 here means a wrong user name or password, not "log in first"
            if (response.StatusCode == System.Net.HttpStatusCode.Unauthorized)
                throw new ApiValidationException(new[] { "User name or password is wrong." });

            await ApiResponseReader.EnsureSuccessAsync(response);

            var login = await response.Content.ReadFromJsonAsync<LoginResponse>(ApiResponseReader.JsonOptions);
            if (login is null || string.IsNullOrWhiteSpace(login.Token))
                throw new ApiRequestException("The server answered, but not with a login. Try again.");

            _token = login.Token;
            _expiresAtUtc = login.ExpiresAtUtc.ToUniversalTime();
            UserName = userName.Trim();

            try
            {
                await SecureStorage.Default.SetAsync(TokenKey, _token);
                await SecureStorage.Default.SetAsync(ExpiresKey, _expiresAtUtc.ToString("O"));
                await SecureStorage.Default.SetAsync(UserNameKey, UserName);
            }
            catch (Exception)
            {
                // couldn't save it for next time. the login still works for this run
            }

            SignInChanged?.Invoke(this, EventArgs.Empty);
        }

        public async Task SignOutAsync()
        {
            await ClearAsync();
        }

        public async Task<bool> RequestLoginAsync()
        {
            // a login that was left open gets cancelled, so nothing waits forever
            _pendingLogin?.TrySetResult(false);

            var pending = new TaskCompletionSource<bool>(TaskCreationOptions.RunContinuationsAsynchronously);
            _pendingLogin = pending;

            await _navigation.GoToAsync(Routes.Login);
            return await pending.Task;
        }

        public void CompleteLogin(bool signedIn)
        {
            _pendingLogin?.TrySetResult(signedIn);
            _pendingLogin = null;
        }

        public async Task<T> RunWriteAsync<T>(Func<Task<T>> write)
        {
            try
            {
                return await write();
            }
            catch (AuthenticationRequiredException)
            {
                // no token, or an old one the server turned down: forget it, ask for a new login
                await ClearAsync();

                if (!await RequestLoginAsync())
                    throw new LoginCancelledException();

                return await write();
            }
        }

        private async Task ClearAsync()
        {
            var wasSignedIn = _token is not null;

            _token = null;
            _expiresAtUtc = DateTime.MinValue;
            UserName = string.Empty;

            try
            {
                SecureStorage.Default.Remove(TokenKey);
                SecureStorage.Default.Remove(ExpiresKey);
                SecureStorage.Default.Remove(UserNameKey);
            }
            catch (Exception)
            {
                // nothing stored, or storage unavailable: either way we're signed out
            }

            if (wasSignedIn)
                SignInChanged?.Invoke(this, EventArgs.Empty);

            await Task.CompletedTask;
        }
    }

    // to every write (POST, PUT, DELETE). reads (GET) go out with no header,
    // because browsing is public. if there's no token yet the write goes out

    public class BearerTokenHandler : DelegatingHandler
    {
        private readonly IAuthService _authService;

        public BearerTokenHandler(IAuthService authService)
        {
            _authService = authService;
        }

        protected override async Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken cancellationToken)
        {
            var isRead = request.Method == HttpMethod.Get || request.Method == HttpMethod.Head;
            if (!isRead)
            {
                var token = await _authService.GetTokenAsync();
                if (token is not null)
                    request.Headers.Authorization = new AuthenticationHeaderValue("Bearer", token);
            }

            return await base.SendAsync(request, cancellationToken);
        }
    }

    public enum GpaStatus
    {
        Safe,
        Watch,
        AtRisk
    }

    public class GpaCalculatorService
    {
        private static readonly Dictionary<string, double> GradeTable = new()
        {
            { "A",  4.0 }, { "A-", 3.7 },
            { "B+", 3.3 }, { "B",  3.0 }, { "B-", 2.7 },
            { "C+", 2.3 }, { "C",  2.0 }, { "C-", 1.7 },
            { "D+", 1.3 }, { "D",  1.0 }, { "D-", 0.7 },
            { "F",  0.0 }
        };

        public double GetGradePoints(string letterGrade)
        {
            if (string.IsNullOrWhiteSpace(letterGrade))
                return 0.0;

            return GradeTable.TryGetValue(letterGrade.Trim().ToUpperInvariant(), out var points)
                ? points
                : 0.0;
        }

        public string GetLetterGradeFromPercent(double percent) => percent switch
        {
            >= 93 => "A",
            >= 90 => "A-",
            >= 87 => "B+",
            >= 83 => "B",
            >= 80 => "B-",
            >= 77 => "C+",
            >= 73 => "C",
            >= 70 => "C-",
            >= 67 => "D+",
            >= 63 => "D",
            >= 60 => "D-",
            _ => "F"
        };

        public double? GetCoursePercent(Course course)
        {
            // OPTIMIZATION: Replaced .Where().ToList() and .Sum() with a single-pass O(N) loop
            // to avoid multiple iterations and intermediate list allocations.
            double earned = 0.0;
            double possible = 0.0;
            int count = 0;

            foreach (var a in course.Assignments)
            {
                if (a.PointsEarned.HasValue && a.PointsPossible > 0)
                {
                    earned += a.PointsEarned.Value;
                    possible += a.PointsPossible;
                    count++;
                }
            }

            if (count == 0 || possible == 0)
                return null;

            return (earned / possible) * 100.0;
        }

        /// <summary>GPA across all courses â€” bigger classes count for more, same as a real transcript.</summary>
        public double CalculateGpa(IEnumerable<Course> courses)
        {
            // OPTIMIZATION: Single-pass O(N) loop avoids memory allocation for intermediate lists.
            double totalPoints = 0.0;
            double totalCredits = 0.0;
            int count = 0;

            foreach (var c in courses)
            {
                if (c.Credits > 0)
                {
                    totalPoints += c.GradePoints * c.Credits;
                    totalCredits += c.Credits;
                    count++;
                }
            }

            if (count == 0 || totalCredits == 0)
                return 0.0;

            return totalPoints / totalCredits;
        }

        /// <summary>Same as above, but also blends in a starting GPA from before this app (past semesters, transfer credits).</summary>
        public double CalculateGpa(IEnumerable<Course> courses, double startingGpa, double startingCredits)
        {
            // OPTIMIZATION: Single-pass O(N) loop avoids memory allocation for intermediate lists.
            double totalPoints = 0.0;
            double totalCredits = 0.0;

            foreach (var c in courses)
            {
                if (c.Credits > 0)
                {
                    totalPoints += c.GradePoints * c.Credits;
                    totalCredits += c.Credits;
                }
            }

            // only blend it in if there's an actual credit count behind it
            if (startingCredits > 0)
            {
                totalPoints += startingGpa * startingCredits;
                totalCredits += startingCredits;
            }

            return totalCredits == 0 ? 0.0 : totalPoints / totalCredits;
        }

        /// <summary>
        /// PTK has two real numbers, not one: 3.5 to get invited, 3.0 to stay
        /// a member in good standing (per MSTC's Beta Chi Theta bylaws).
        /// Safe = at/above the invite GPA. Watch = a member in good standing
        /// but below the invite bar. AtRisk = below good standing, membership
        /// is actually on the line.
        /// </summary>
        public GpaStatus GetStatus(double currentGpa, double goodStandingGpa, double initiationGpa)
        {
            if (currentGpa >= initiationGpa)
                return GpaStatus.Safe;

            if (currentGpa >= goodStandingGpa)
                return GpaStatus.Watch;

            return GpaStatus.AtRisk;
        }
    }

                _courses[index] = course;
            else
                _courses.Add(course);

            return Task.FromResult(course);
        }

namespace GritTrack.Api.Configuration;

/// <summary>
/// Reads the repo's .env file (the same one docker compose uses) so the API and the
/// SQL container share one password. Real environment variables always win over the file.
/// </summary>
public static class DotEnvLoader
{
    /// <summary>Looks for .env in the start folder and then in each parent folder, and loads the first one it finds.</summary>
    public static void Load(string startFolder)
    {
        var folder = new DirectoryInfo(startFolder);
        while (folder is not null)
        {
            var filePath = Path.Combine(folder.FullName, ".env");
            if (File.Exists(filePath))
            {
                LoadFile(filePath);
                return;
            }

            folder = folder.Parent;
        }
    }

    /// <summary>Sets one environment variable for every NAME=value line, skipping blanks and # comments.</summary>
    private static void LoadFile(string filePath)
    {
        foreach (var rawLine in File.ReadAllLines(filePath))
        {
            var line = rawLine.Trim();
            if (line.Length == 0 || line.StartsWith('#'))
                continue;

            var separatorIndex = line.IndexOf('=');
            if (separatorIndex <= 0)
                continue;

            var name = line[..separatorIndex].Trim();
            var value = line[(separatorIndex + 1)..].Trim().Trim('"');

            // a variable that is already set in the real environment wins
            if (Environment.GetEnvironmentVariable(name) is null)
                Environment.SetEnvironmentVariable(name, value);
        }
    }
}

/// <summary>
/// Builds the SQL Server connection string. Server and database name come from appsettings.json,
/// the password only ever comes from the environment (.env) or user secrets, never from a file in git.
/// </summary>
public static class SqlConnectionStringFactory
{
    /// <summary>Where the password lives: the MSSQL_SA_PASSWORD environment variable or user secret.</summary>
    public const string PasswordKey = "MSSQL_SA_PASSWORD";

    /// <summary>Builds the connection string, or stops with a clear message if the password is missing.</summary>
    public static string Build(IConfiguration configuration)
    {
        var password = configuration[PasswordKey];
        if (string.IsNullOrWhiteSpace(password))
        {
            throw new InvalidOperationException(
                $"DataSource is SqlServer but {PasswordKey} is not set. " +
                "Copy .env.example to .env and set it there (or use dotnet user-secrets).");
        }

        return new SqlConnectionStringBuilder
        {
            DataSource = configuration["Database:Server"] ?? "127.0.0.1,1433",
            InitialCatalog = configuration["Database:Name"] ?? "GritTrack",
            UserID = "sa",
            Password = password,
            Encrypt = true,
            // the dev container uses a self-signed certificate
            TrustServerCertificate = true
        }.ConnectionString;
    }

public class AuthController : ControllerBase
{
    private readonly IJwtTokenService _tokenService;
    private readonly AuthOptions _options;

    private readonly ILogger<AuthController> _logger;

    public AuthController(IJwtTokenService tokenService, IOptions<AuthOptions> options, ILogger<AuthController> logger)
    {
        _tokenService = tokenService;
        _options = options.Value;
        _logger = logger;
    }

    /// <summary>POST /api/auth/login. 200 with a token, 400 if a field is missing, 401 for a wrong user name or password.</summary>
    [HttpPost("login")]
    public ActionResult<LoginResponseDto> Login(LoginRequestDto request)
    {
        // with auth off there is nothing to log in to
        if (!_options.Enabled)
        {
            return Problem(
                statusCode: StatusCodes.Status404NotFound,
                title: "Authentication is turned off",
                detail: "Writes do not need a token while Auth:Enabled is false.");
        }

        if (!_tokenService.AreCredentialsValid(request.UserName, request.Password))
        {
            // log that it happened, never what was typed: no user name, no password
            _logger.LogWarning("Login failed");

            // same message for a wrong user name and a wrong password on purpose
            return Problem(
                statusCode: StatusCodes.Status401Unauthorized,
                title: "Login failed",
                detail: "User name or password is wrong.");
        }

        return Ok(_tokenService.CreateToken(request.UserName));
    }
}

[ApiController]

public class CoursesController : ControllerBase

    }

    /// <summary>GET /api/courses, optionally narrowed with ?grade=.</summary>
    [HttpGet]
    public async Task<ActionResult<IEnumerable<CourseResponseDto>>> GetAll(
        [FromQuery] string? grade, CancellationToken cancellationToken)
    {
        var courses = await _service.GetCoursesAsync(grade, cancellationToken);
        return Ok(courses.Select(ToResponseDto));
    }

    /// <summary>GET /api/courses/{id}.</summary>
    [HttpGet("{id:int}", Name = nameof(GetById))]
    public async Task<ActionResult<CourseResponseDto>> GetById(int id, CancellationToken cancellationToken)
    {
        var course = await _service.GetCourseByIdAsync(id, cancellationToken);
        return course is null ? NotFound() : Ok(ToResponseDto(course));
    }

    /// <summary>POST /api/courses. 401 without a token (when auth is on), 409 if the course code is already taken.</summary>
    [HttpPost]
    [Authorize(Policy = AuthPolicies.WriteAccess)]
    public async Task<ActionResult<CourseResponseDto>> Create(
        CourseRequestDto request, CancellationToken cancellationToken)
    {
        var (outcome, course) = await _service.CreateCourseAsync(request, cancellationToken);

        return outcome switch
        {
            SaveCourseOutcome.DuplicateCourseCode => DuplicateCourseCodeProblem(request.CourseCode),
            SaveCourseOutcome.Created => CreatedAtAction(nameof(GetById), new { id = course!.Id }, ToResponseDto(course)),
            _ => throw new InvalidOperationException($"Unexpected create outcome: {outcome}.")
        };
    }

    /// <summary>PUT /api/courses/{id}. 404s if it doesn't exist, 409s if the new code is already taken by a different course.</summary>
    [HttpPut("{id:int}")]
    [Authorize(Policy = AuthPolicies.WriteAccess)]
    public async Task<ActionResult<CourseResponseDto>> Update(
        int id, CourseRequestDto request, CancellationToken cancellationToken)
    {
        var (outcome, course) = await _service.UpdateCourseAsync(id, request, cancellationToken);

        return outcome switch
        {
            SaveCourseOutcome.NotFound => NotFound(),
            SaveCourseOutcome.DuplicateCourseCode => DuplicateCourseCodeProblem(request.CourseCode),
            SaveCourseOutcome.Updated => Ok(ToResponseDto(course!)),
            _ => throw new InvalidOperationException($"Unexpected update outcome: {outcome}.")
        };
    }

    /// <summary>DELETE /api/courses/{id}.</summary>
    [HttpDelete("{id:int}")]
    [Authorize(Policy = AuthPolicies.WriteAccess)]
    public async Task<IActionResult> Delete(int id, CancellationToken cancellationToken)
    {
        var deleted = await _service.DeleteCourseAsync(id, cancellationToken);

    }

using GritTrack.Api.Models;

namespace GritTrack.Api.Repositories;

/// <summary>
/// Stores courses in Data/courses.json. One semaphore guards every method
/// that touches the file, so two requests can never read or write it at
/// the same time and corrupt it.
/// </summary>
public class JsonCourseRepository : ICourseRepository
{
    private readonly string _filePath;
    private readonly SemaphoreSlim _fileLock = new(1, 1);

    private static readonly JsonSerializerOptions SerializerOptions = new()
    {
        WriteIndented = true
    };

    public JsonCourseRepository(IWebHostEnvironment environment)
    {
        var dataFolder = Path.Combine(environment.ContentRootPath, "Data");
        Directory.CreateDirectory(dataFolder);
        _filePath = Path.Combine(dataFolder, "courses.json");
    }

    /// <inheritdoc />
    public async Task<List<Course>> GetAllAsync(CancellationToken cancellationToken)
    {
        await _fileLock.WaitAsync(cancellationToken);
        try
        {
            return await ReadAllAsync(cancellationToken);
        }
        finally
        {
            _fileLock.Release();
        }
    }

    /// <inheritdoc />
    public async Task<Course?> GetByIdAsync(int id, CancellationToken cancellationToken)
    {
        await _fileLock.WaitAsync(cancellationToken);
        try
        {
            var courses = await ReadAllAsync(cancellationToken);
            return courses.FirstOrDefault(c => c.Id == id);
        }
        finally
        {
            _fileLock.Release();
        }
    }

    /// <inheritdoc />
    public async Task<Course?> GetByCourseCodeAsync(string courseCode, CancellationToken cancellationToken)
    {
        await _fileLock.WaitAsync(cancellationToken);
        try
        {
            var courses = await ReadAllAsync(cancellationToken);
            return courses.FirstOrDefault(c =>
                string.Equals(c.CourseCode, courseCode, StringComparison.OrdinalIgnoreCase));
        }
        finally
        {
            _fileLock.Release();
        }
    }

    /// <inheritdoc />
    public async Task<Course> AddAsync(Course course, CancellationToken cancellationToken)
    {
        await _fileLock.WaitAsync(cancellationToken);
        try
        {
            var courses = await ReadAllAsync(cancellationToken);

            // the repo hands out the id, not the caller â€” keeps two
            // requests from ever landing on the same one
            course.Id = courses.Count == 0 ? 1 : courses.Max(c => c.Id) + 1;
            courses.Add(course);

            await WriteAllAsync(courses, cancellationToken);
            return course;
        }
        finally
        {
            _fileLock.Release();
        }
    }

    /// <inheritdoc />
    public async Task<bool> UpdateAsync(Course course, CancellationToken cancellationToken)
    {
        await _fileLock.WaitAsync(cancellationToken);
        try
        {
            var courses = await ReadAllAsync(cancellationToken);
            var index = courses.FindIndex(c => c.Id == course.Id);
            if (index < 0)
                return false;

            courses[index] = course;
            await WriteAllAsync(courses, cancellationToken);
            return true;
        }
        finally
        {
            _fileLock.Release();
        }
    }

    /// <inheritdoc />
    public async Task<bool> DeleteAsync(int id, CancellationToken cancellationToken)
    {
        await _fileLock.WaitAsync(cancellationToken);
        try
        {
            var courses = await ReadAllAsync(cancellationToken);
            var removed = courses.RemoveAll(c => c.Id == id);
            if (removed == 0)
                return false;

            await WriteAllAsync(courses, cancellationToken);
            return true;
        }
        finally
        {
            _fileLock.Release();
        }
    }

    /// <summary>Reads the whole file. Caller already holds _fileLock before this runs.</summary>
    private async Task<List<Course>> ReadAllAsync(CancellationToken cancellationToken)
    {
        if (!File.Exists(_filePath))
            return new List<Course>();

        await using var stream = File.OpenRead(_filePath);
        if (stream.Length == 0)
            return new List<Course>();

        var courses = await JsonSerializer.DeserializeAsync<List<Course>>(stream, SerializerOptions, cancellationToken);
        return courses ?? new List<Course>();
    }

    /// <summary>
    /// Writes to a temp file first, then swaps it in â€” a crash mid-write
    /// can't leave a half-written courses.json behind. Caller already
    /// holds _fileLock before this runs.
    /// </summary>
    private async Task WriteAllAsync(List<Course> courses, CancellationToken cancellationToken)
    {
        var tempPath = _filePath + ".tmp";
        await using (var stream = File.Create(tempPath))
        {
            await JsonSerializer.SerializeAsync(stream, courses, SerializerOptions, cancellationToken);
        }

        File.Move(tempPath, _filePath, overwrite: true);
    }
}

            return false;

        // when the caller already edited the tracked row itself there is nothing to copy
        if (!ReferenceEquals(stored, course))
            _context.Entry(stored).CurrentValues.SetValues(course);

        await _context.SaveChangesAsync(cancellationToken);
        return true;
    }

    {
        var stored = await _context.Courses.FirstOrDefaultAsync(c => c.Id == id, cancellationToken);
        if (stored is null)
            return false;

        _context.Courses.Remove(stored);
        await _context.SaveChangesAsync(cancellationToken);
        return true;

    // is worth the same GPA points no matter which one touched it
    private static readonly Dictionary<string, double> GradeTable = new(StringComparer.OrdinalIgnoreCase)
    {
        { "A", 4.0 }, { "A-", 3.7 },
        { "B+", 3.3 }, { "B", 3.0 }, { "B-", 2.7 },

        { "D+", 1.3 }, { "D", 1.0 }, { "D-", 0.7 },
        { "F", 0.0 }
    };

    private readonly ILogger<CourseService> _logger;

    public CourseService(ICourseRepository repository, ILogger<CourseService> logger)
    {
        _repository = repository;
        _logger = logger;
    }

    /// <inheritdoc />
    public async Task<List<Course>> GetCoursesAsync(string? grade, CancellationToken cancellationToken)
    {
        var courses = await _repository.GetAllAsync(cancellationToken);

        if (!string.IsNullOrWhiteSpace(grade))
        {
            courses = courses
                .Where(c => string.Equals(c.CurrentGrade, grade, StringComparison.OrdinalIgnoreCase))
                .ToList();
        }

        return courses;
    }

    /// <inheritdoc />
    public Task<Course?> GetCourseByIdAsync(int id, CancellationToken cancellationToken)
        => _repository.GetByIdAsync(id, cancellationToken);

    /// <inheritdoc />
    public async Task<(SaveCourseOutcome Outcome, Course? Course)> CreateCourseAsync(CourseRequestDto request, CancellationToken cancellationToken)
    {
        // the business rule: two courses can't share a course code
        var existing = await _repository.GetByCourseCodeAsync(request.CourseCode, cancellationToken);
        if (existing is not null)
            return (SaveCourseOutcome.DuplicateCourseCode, null);

        var course = new Course
        {
            Name = request.Name.Trim(),
            CourseCode = request.CourseCode.Trim(),
            Instructor = request.Instructor?.Trim(),
            Credits = request.Credits,
            CurrentGrade = request.CurrentGrade.Trim(),
            GradePoints = GetGradePoints(request.CurrentGrade),
            CreatedAt = DateTime.UtcNow
        };

        var saved = await _repository.AddAsync(course, cancellationToken);

        // one line per write, and only the id and code: nothing else about the course or the caller
        _logger.LogInformation("Course created: id {CourseId}, code {CourseCode}", saved.Id, ForLog(saved.CourseCode));
        return (SaveCourseOutcome.Created, saved);
    }

    /// <inheritdoc />
    public async Task<(SaveCourseOutcome Outcome, Course? Course)> UpdateCourseAsync(int id, CourseRequestDto request, CancellationToken cancellationToken)
    {
        var existing = await _repository.GetByIdAsync(id, cancellationToken);
        if (existing is null)
            return (SaveCourseOutcome.NotFound, null);

        // only a problem if some OTHER course already owns this code
        var codeOwner = await _repository.GetByCourseCodeAsync(request.CourseCode, cancellationToken);
        if (codeOwner is not null && codeOwner.Id != id)
            return (SaveCourseOutcome.DuplicateCourseCode, null);

        existing.Name = request.Name.Trim();
        existing.CourseCode = request.CourseCode.Trim();
        existing.Instructor = request.Instructor?.Trim();
        existing.Credits = request.Credits;
        existing.CurrentGrade = request.CurrentGrade.Trim();
        existing.GradePoints = GetGradePoints(request.CurrentGrade);

        await _repository.UpdateAsync(existing, cancellationToken);

        _logger.LogInformation("Course updated: id {CourseId}, code {CourseCode}", existing.Id, ForLog(existing.CourseCode));
        return (SaveCourseOutcome.Updated, existing);
    }

    /// <inheritdoc />
    public async Task<bool> DeleteCourseAsync(int id, CancellationToken cancellationToken)
    {
        // look the course up first, so the log line can say which code was deleted
        var existing = await _repository.GetByIdAsync(id, cancellationToken);
        if (existing is null)
            return false;

        var code = existing.CourseCode;
        var deleted = await _repository.DeleteAsync(id, cancellationToken);
        if (deleted)
            _logger.LogInformation("Course deleted: id {CourseId}, code {CourseCode}", id, ForLog(code));

        return deleted;
    }

    /// <summary>Strips line breaks and other control characters, so a made-up course code can't fake extra log lines.</summary>
    private static string ForLog(string text)
        => new(text.Where(character => !char.IsControl(character)).ToArray());

    /// <summary>Looks up how many GPA points a letter grade is worth. Unknown grades are worth 0.</summary>
    private static double GetGradePoints(string letterGrade)
        => GradeTable.TryGetValue(letterGrade.Trim(), out var points) ? points : 0.0;
}

/// <inheritdoc cref="IJwtTokenService" />
public class JwtTokenService : IJwtTokenService
{
    private readonly AuthOptions _options;

    public JwtTokenService(IOptions<AuthOptions> options)
    {
        _options = options.Value;
    }

    /// <inheritdoc />
    public bool AreCredentialsValid(string userName, string password)
    {
        // both checks always run and compare in fixed time, so a wrong guess
        // can't be told apart from a right one by how long the answer took
        var userNameMatches = FixedTimeTextEquals(userName, _options.DemoUserName);
        var passwordMatches = FixedTimeTextEquals(password, _options.DemoPassword ?? string.Empty);
        return userNameMatches & passwordMatches;
    }

    /// <inheritdoc />
    public LoginResponseDto CreateToken(string userName)
    {
        var expiresAtUtc = DateTime.UtcNow.AddMinutes(_options.TokenMinutes);
        var signingKey = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(_options.SigningKey!));

        var descriptor = new SecurityTokenDescriptor
        {
            Issuer = _options.Issuer,
            Audience = _options.Audience,
            Subject = new ClaimsIdentity([new Claim(ClaimTypes.Name, userName)]),
            Expires = expiresAtUtc,
            SigningCredentials = new SigningCredentials(signingKey, SecurityAlgorithms.HmacSha256)
        };

        return new LoginResponseDto
        {
            Token = new JsonWebTokenHandler().CreateToken(descriptor),
            ExpiresAtUtc = expiresAtUtc
        };
    }

    /// <summary>Compares two strings by their hashes, so the comparison time doesn't depend on where they differ.</summary>
    private static bool FixedTimeTextEquals(string first, string second)
        => CryptographicOperations.FixedTimeEquals(
            SHA256.HashData(Encoding.UTF8.GetBytes(first)),
            SHA256.HashData(Encoding.UTF8.GetBytes(second)));

    // same policy name, but everyone passes, so the controller doesn't need to know which mode we're in
    builder.Services.AddAuthorizationBuilder()
        .AddPolicy(AuthPolicies.WriteAccess, policy => policy.RequireAssertion(_ => true));
}

var app = builder.Build();

// creates or updates the tables before the first request
if (dataSource == DataSources.SqlServer)
    await ApplyMigrationsAsync(app);

// one place that turns an unhandled exception into a proper ProblemDetails
// response, instead of a try/catch in every action
app.UseExceptionHandler();

if (app.Environment.IsDevelopment())
{
    app.MapOpenApi();
}

app.UseHttpsRedirection();

// no token or a bad one on a protected action = 401
if (authOptions.Enabled)
    app.UseAuthentication();
app.UseAuthorization();

app.MapControllers();

// the two settings that change how the app behaves, stated once at startup.
// (only the names: never the connection string, password or signing key)
app.Logger.LogInformation("Data source: {DataSource}", dataSource);
app.Logger.LogInformation("Authentication: {AuthState}", authOptions.Enabled ? "ON (writes need a token)" : "OFF (writes are open)");

app.Run();

/// <summary>
/// Refuses to start with auth on but the secrets missing. The secrets live in dotnet user-secrets
/// (only loaded when the environment is Development), never in a file that goes to git.
/// </summary>
static void ValidateAuthSettings(AuthOptions authOptions)
{
    const int minimumSigningKeyLength = 32;
    var problems = new List<string>();

    if (string.IsNullOrWhiteSpace(authOptions.SigningKey) || authOptions.SigningKey.Length < minimumSigningKeyLength)
        problems.Add($"Auth:SigningKey is missing or shorter than {minimumSigningKeyLength} characters");
    if (string.IsNullOrWhiteSpace(authOptions.DemoPassword))
        problems.Add("Auth:DemoPassword is missing");

    if (problems.Count == 0)
        return;

    throw new InvalidOperationException(
        $"Auth:Enabled is true but {string.Join(" and ", problems)}. " +
        "Set them with: dotnet user-secrets set \"Auth:SigningKey\" \"<long random text>\" " +
        "and dotnet user-secrets set \"Auth:DemoPassword\" \"<your demo password>\" " +
        "(run from the GritTrack.Api folder), or set Auth:Enabled to false.");
}

/// <summary>
/// Creates or updates the database tables when the app starts. SQL Server can take a few
/// seconds after the container says "up", so this tries a few times before giving up.
/// </summary>
static async Task ApplyMigrationsAsync(WebApplication app)
{
    const int maximumAttempts = 6;

    for (var attempt = 1; ; attempt++)
    {
        try
        {
            using var scope = app.Services.CreateScope();
            var context = scope.ServiceProvider.GetRequiredService<GritTrackDbContext>();
            await context.Database.MigrateAsync();
            return;
        }
        catch (Exception exception) when (attempt < maximumAttempts)
        {
            app.Logger.LogWarning("Database not ready yet (attempt {Attempt} of {Maximum}): {Reason}",
                attempt, maximumAttempts, exception.GetType().Name);
            await Task.Delay(TimeSpan.FromSeconds(5));
        }
    }
}
